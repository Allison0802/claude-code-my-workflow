# `method-evolve` Skill — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-project Claude Code skill that runs an evolutionary code-search loop over statistical estimators, grounded in the Round-2-reviewed specification for the Missing Types / IPW propensity feature-set search.

**Architecture:** Two-layer — a domain-agnostic skill at `.claude/skills/method-evolve/` (for `slot.kind = feature_set` only, per §3 tightened claim); a per-project `config.yaml` generated by `scan` + `setup` that declares the grammar, fitness, gates, evaluator wiring, baselines, and compute budget. Runs split local Screen (in-session Claude Code) / Longleaf Full (user-run `sbatch`).

**Tech Stack:** R 4.4.0, `testthat`, `yaml`, `jsonlite`, `rlang`, `digest`, `parallel::mclapply`, `tools::pskill`, `flock` (via `filelock` R package). Skill Markdown + Bash. Task tool subagent (Claude Opus, explicit).

**Spec reference:** [`quality_reports/specs/2026-04-21_method-evolve-design-v2.md`](../specs/2026-04-21_method-evolve-design-v2.md) (the Round-2-approved 720-line spec). Every task below cites its spec section.

**Reference paper:** Tsinghua FIB Lab, *AutoSOTA: An End-to-End Automated Research System for SOTA Model Discovery* (2026) — <https://tsinghua-fib-lab.github.io/AutoSOTA/AutoSOTA.pdf>. The skill borrows AutoSOTA's stage-3 evolutionary code-search loop (program DB → sampler → LLM ensemble → evaluator pool → update DB) and its Red-Line-Supervisor pattern (here: hard statistical gates). It does **not** implement AutoSOTA's stages 1–2 (paper-to-repo grounding, automated rubric construction), which are outside v1 scope (see non-goals §2 of the spec). Domain-specific adaptations in this plan — typed R-grammar validator, pseudo-Brier recovery fitness, dual-tier coverage gates, scenario-matched baselines — are required because Monte Carlo evaluators for statistical estimators need stricter correctness guardrails than AutoSOTA's benchmark-driven targets.

**Project conventions:** `here::here()`, `set.seed(YYYYMMDD)`, Okabe-Ito palette, dated subfolders with `-NN` collision suffix, session logging under `.claude/rules/session-logging.md`, skill output storage per `CLAUDE.md`.

```r
parse_cli_args <- function(argv) {
  if (length(argv) < 1L) me_stop("Usage: cli.R <command> [--key=value ...]")
  cmd  <- argv[[1L]]; rest <- argv[-1L]; opts <- list()
  for (a in rest) {
    if (!grepl("^--[^=]+=", a)) me_stop("Bad flag: %s (need --key=value)", a)
    kv <- sub("^--", "", a)
    opts[[sub("=.*$", "", kv)]] <- sub("^[^=]+=", "", kv)
  }
  list(command = cmd, opts = opts)
}

main <- function() {
  args <- commandArgs(trailingOnly = TRUE)

    version = cat(sprintf("method-evolve v%s\n", ME_VERSION)),
    me_stop("Unknown command: %s", p$command)
  )
}
```

```r
# ibs-patch-version: 1  (method-evolve prereq)
# ============================================================================
# Pseudo-Brier helpers, sourced by missing_types_method_comparison_*.R scripts
# ============================================================================

pseudo_brier <- function(pseudo_y, phat) {
  stopifnot(length(pseudo_y) == length(phat))
  mean((pseudo_y - phat)^2)
}

integrate_brier <- function(time_grid, brier_t) {
  if (length(time_grid) != length(brier_t))
    stop("time_grid and brier_t must have the same length")
  if (length(time_grid) < 2L) return(brier_t[[1L]])
  dx  <- diff(time_grid)
  avg <- (brier_t[-length(brier_t)] + brier_t[-1L]) / 2
  sum(dx * avg) / (time_grid[length(time_grid)] - time_grid[1L])
}

compute_pseudo_brier <- function(time_grid, pseudo_obs_mat, pred_surv_mat) {
  # rows = subjects, cols = grid points (parallel)
  stopifnot(ncol(pseudo_obs_mat) == length(time_grid),
            dim(pseudo_obs_mat) == dim(pred_surv_mat))
  brier_t <- vapply(seq_along(time_grid), function(j) {
    pseudo_brier(pseudo_obs_mat[, j], pred_surv_mat[, j])
  }, numeric(1))
  integrate_brier(time_grid, brier_t)
}
```

```r
# Phase -1: prerequisites. Owns the IBS patch and (T2) baseline meta sidecars.

IBS_PATCH_VERSION <- 1L
IBS_STAMP_RE <- "# ibs-patch-version:\\s*([0-9]+)"

has_ibs_patch <- function(path, required_version = IBS_PATCH_VERSION) {
  if (!file.exists(path)) return(FALSE)
  lines <- readLines(path, warn = FALSE)
  m <- regmatches(lines, regexec(IBS_STAMP_RE, lines))
  found <- Filter(function(x) length(x) == 2L, m)
  if (!length(found)) return(FALSE)
  ver <- as.integer(found[[1L]][[2L]])
  isTRUE(ver >= required_version)
}

apply_ibs_patch <- function(target_path, patch_path) {
  if (has_ibs_patch(target_path)) {
    me_log("INFO", "IBS patch already present (version ≥ %d) — skipping", IBS_PATCH_VERSION)
    return(invisible(FALSE))
  }
  patch_lines <- readLines(patch_path, warn = FALSE)
  target_lines <- readLines(target_path, warn = FALSE)
  writeLines(c(patch_lines, "", target_lines), target_path)
  me_log("INFO", "Applied IBS patch v%d to %s", IBS_PATCH_VERSION, target_path)
  invisible(TRUE)
}
```

```r
BASELINE_HARD_KEYS <- c("scenario_name","n_subjects","n_sims",
                        "dgp_version","ibs_patch_version")
BASELINE_WARN_KEYS <- c("evaluator_sha")

read_meta_sidecar <- function(path) {
  if (!file.exists(path)) return(NULL)
  jsonlite::fromJSON(paste(readLines(path, warn = FALSE), collapse = "\n"),
                    simplifyVector = TRUE)
}

check_baseline_meta <- function(meta_path, expected) {
  m <- read_meta_sidecar(meta_path)
  if (is.null(m)) return(list(ok = FALSE,
                               mismatches = "meta sidecar missing",
                               warnings = character()))
  mismatches <- character(); warns <- character()
  for (k in BASELINE_HARD_KEYS) {
    if (!is.null(expected[[k]]) && !identical(m[[k]], expected[[k]])) {
      mismatches <- c(mismatches, sprintf("%s: expected=%s observed=%s",
                                           k, toString(expected[[k]]),
                                           toString(m[[k]])))
    }
  }
  for (k in BASELINE_WARN_KEYS) {
    if (!is.null(expected[[k]]) && !identical(m[[k]], expected[[k]])) {
      warns <- c(warns, sprintf("%s drifted: expected=%s observed=%s",
                                 k, expected[[k]], m[[k]]))
    }
  }
  list(ok = length(mismatches) == 0L, mismatches = mismatches, warnings = warns)
}

write_meta_sidecar <- function(rds_path, scenario_name, n_subjects, n_sims,
                                dgp_version, evaluator_sha, ibs_patch_version) {
  sidecar <- sub("\\.rds$", ".meta.json", rds_path)
  meta <- list(scenario_name = scenario_name,
                n_subjects = as.integer(n_subjects),
                n_sims = as.integer(n_sims),
                dgp_version = dgp_version,
                evaluator_sha = evaluator_sha,
                ibs_patch_version = as.integer(ibs_patch_version),
                created_at = format(Sys.time(), "%FT%T%z"))
  writeLines(jsonlite::toJSON(meta, auto_unbox = TRUE, pretty = TRUE), sidecar)
  invisible(sidecar)
}
```

```r
suppressPackageStartupMessages({ library(yaml) })

ME_VALID_SLOT_KINDS   <- c("feature_set")
ME_VALID_INTERACTIONS <- c("none","pairwise","three-way")
ME_VALID_BACKENDS     <- c("local","longleaf")

load_config <- function(path) {
  if (!file.exists(path)) me_stop("Config file not found: %s", path)
  yaml::read_yaml(path)
}

save_config <- function(cfg, path) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  yaml::write_yaml(cfg, path)
}

validate_config <- function(cfg) {
  req <- c("name","target_script","prerequisites","evaluator","substitutions",
           "slot","fitness","baselines","budget","compute","paths","proposer")
  miss <- setdiff(req, names(cfg))
  if (length(miss)) me_stop("Missing required config keys: %s",
                            paste(miss, collapse = ", "))

  if (!cfg$slot$kind %in% ME_VALID_SLOT_KINDS) {
    me_stop("slot.kind must be one of [%s]",
            paste(ME_VALID_SLOT_KINDS, collapse=", "))
  }
  if (!cfg$slot$interactions %in% ME_VALID_INTERACTIONS) {
    me_stop("slot.interactions must be one of [%s]",
            paste(ME_VALID_INTERACTIONS, collapse=", "))
  }
  for (t in cfg$slot$transforms) {
    if (is.null(t$head)) me_stop("transform missing 'head': %s", toString(t$name))
    if (is.null(t$name)) me_stop("transform missing 'name'")
  }

  covg_tiers <- vapply(Filter(function(g) g$metric == "coverage", cfg$fitness$gates),
                        function(g) g$tier, character(1))
  if (!all(c("screen","full") %in% covg_tiers)) {
    me_stop("fitness.gates must include both screen and full coverage gates")
  }

  if (!cfg$compute$screen$backend %in% ME_VALID_BACKENDS)
    me_stop("compute.screen.backend invalid")
  if (!is.numeric(cfg$compute$screen$n_workers) ||
      cfg$compute$screen$n_workers < 1)
    me_stop("compute.screen.n_workers must be positive integer")

  invisible(TRUE)
}

freeze_substitutions <- function(cfg, run_date = format(Sys.Date(), "%Y-%m-%d"),
                                  refreeze = FALSE) {
  frozen <- cfg$substitutions$frozen %||% list()
  if (length(frozen) && !refreeze) {
    me_stop("Substitutions already frozen (run_date=%s). Pass refreeze=TRUE.",
            frozen$run_date %||% "?")
  }
  cfg$substitutions$frozen <- list(run_date = run_date, name = cfg$name)
  cfg
}

resolve_frozen_paths <- function(cfg) {
  frozen <- cfg$substitutions$frozen
  if (!length(frozen)) me_stop("Config not frozen — call freeze_substitutions first")
  sub_in <- function(x) {
    if (!is.character(x)) return(x)
    for (k in names(frozen)) {
      x <- gsub(paste0("\\$\\{", k, "\\}"), frozen[[k]], x, fixed = FALSE)
    }
    # resolve ${out_dir} after out_dir itself is resolved
    if (!is.null(frozen$.out_dir)) {
      x <- gsub("\\$\\{out_dir\\}", frozen$.out_dir, x, fixed = FALSE)
    }
    x
  }
  cfg$paths$out_dir <- sub_in(cfg$paths$out_dir)
  cfg$substitutions$frozen$.out_dir <- cfg$paths$out_dir
  for (p in c("variants_dir","program_db","state_file")) {
    cfg$paths[[p]] <- sub_in(cfg$paths[[p]])
    cfg$paths[[p]] <- gsub("\\$\\{out_dir\\}", cfg$paths$out_dir, cfg$paths[[p]], fixed = FALSE)
  }
  cfg$compute$screen$variants_cache <- sub_in(cfg$compute$screen$variants_cache)
  cfg$compute$full$promote$wrapper_script <- sub_in(cfg$compute$full$promote$wrapper_script)
  cfg$compute$full$promote$manifest_file <- sub_in(cfg$compute$full$promote$manifest_file)
  cfg$compute$full$promote$manifest_file <- gsub("\\$\\{out_dir\\}",
    cfg$paths$out_dir, cfg$compute$full$promote$manifest_file)
  cfg
}

resolve_per_variant <- function(env_vars, variant_path, variant_id, seed_block) {
  subs <- list(VARIANT_PATH = variant_path, VARIANT_ID = variant_id,
                SEED_BLOCK = as.character(seed_block))
  lapply(env_vars, function(v) {
    s <- as.character(v)
    for (k in names(subs)) s <- gsub(paste0("\\$\\{", k, "\\}"), subs[[k]], s, fixed = FALSE)
    s
  })
}
```

```r
validate_feature_set <- function(feature_set, slot) {
  errors <- character()

  if (length(feature_set) > slot$max_terms) {
    errors <- c(errors, sprintf("exceeds max_terms: %d > %d",
                                 length(feature_set), slot$max_terms))
  }

  for (i in seq_along(feature_set)) {
    for (pat in slot$forbidden_patterns) {
      if (grepl(pat, feature_set[[i]])) {
        errors <- c(errors, sprintf("term %d ('%s'): forbidden pattern '%s'",
                                     i, feature_set[[i]], pat))
      }
    }
  }

  descriptors <- slot$transforms
  walker <- .make_walker(slot$whitelist, descriptors, slot$interactions)

  for (i in seq_along(feature_set)) {
    term <- feature_set[[i]]
    e <- tryCatch(rlang::parse_expr(term),
                  error = function(err) structure(list(msg = conditionMessage(err)),
                                                   class = "me_parse_err"))
    if (inherits(e, "me_parse_err")) {
      errors <- c(errors, sprintf("term %d ('%s'): parse_error: %s",
                                   i, term, e$msg))
      next
    }
    r <- walker(e)
    if (!r$ok) {
      errors <- c(errors, sprintf("term %d ('%s'): %s", i, term, r$reason))
    }
  }

  list(valid = length(errors) == 0L, errors = errors)
}

.make_walker <- function(whitelist, descriptors, interactions) {
  .check_base <- function(sym) {
    nm <- as.character(sym)
    if (!nm %in% whitelist) return(list(ok = FALSE,
      reason = sprintf("base '%s' not in whitelist", nm)))
    list(ok = TRUE)
  }
  .check_transform <- function(e) {
    head <- as.character(e[[1L]])
    cands <- Filter(function(d) d$head == head, descriptors)
    if (!length(cands)) return(list(ok = FALSE,
      reason = sprintf("transform '%s' not in descriptors", head)))
    # arity + kwargs matching
    args <- as.list(e)[-1L]
    # first positional arg must be a whitelisted base (for simple unary transforms)
    # for I(<base>^2) shape, first arg is an expression of form `<base>^2`
    for (d in cands) {
      if (!is.null(d$shape) && d$shape == "I(<base>^2)") {
        if (length(args) != 1L) next
        inner <- args[[1L]]
        if (is.call(inner) && as.character(inner[[1L]]) == "^" &&
            is.name(inner[[2L]]) && identical(inner[[3L]], 2)) {
          bc <- .check_base(inner[[2L]])
          if (bc$ok) return(list(ok = TRUE))
        }
        next
      }
      if (is.null(d$kwargs)) {
        if (length(args) == 1L && is.name(args[[1L]])) {
          bc <- .check_base(args[[1L]])
          if (bc$ok) return(list(ok = TRUE))
        }
      } else {
        # first positional + kwargs
        if (length(args) < 1L || !is.name(args[[1L]])) next
        bc <- .check_base(args[[1L]])
        if (!bc$ok) next
        # collect named args
        nms <- names(args)
        named <- if (is.null(nms)) list() else args[nzchar(nms)]
        # evaluate simple numeric literals
        named_vals <- lapply(named, function(a) tryCatch(eval(a), error = function(e) NULL))
        if (identical(named_vals[names(d$kwargs)], d$kwargs)) {
          return(list(ok = TRUE))
        }
      }
    }
    list(ok = FALSE,
         reason = sprintf("ns_args_mismatch or kwargs mismatch for '%s'", head))
  }
  .check_interaction <- function(e) {
    parts <- .flatten_interaction(e)
    arity <- length(parts)
    if (interactions == "none") return(list(ok = FALSE, reason = "interactions disabled"))
    if (interactions == "pairwise" && arity > 2L)
      return(list(ok = FALSE, reason = sprintf("three-way interaction under pairwise arity=%d", arity)))
    for (p in parts) {
      if (!is.name(p)) return(list(ok = FALSE, reason = "interaction part not a bare base"))
      bc <- .check_base(p); if (!bc$ok) return(bc)
    }
    list(ok = TRUE)
  }
  .flatten_interaction <- function(e) {
    if (is.call(e) && as.character(e[[1L]]) %in% c("*",":")) {
      c(.flatten_interaction(e[[2L]]), .flatten_interaction(e[[3L]]))
    } else list(e)
  }
  function(e) {
    if (is.name(e)) return(.check_base(e))
    if (is.call(e)) {
      head <- as.character(e[[1L]])
      if (head %in% c("*",":")) return(.check_interaction(e))
      return(.check_transform(e))
    }
    list(ok = FALSE, reason = "unsupported expression form")
  }
}

feature_set_to_formula <- function(feature_set) {
  paste(feature_set, collapse = " + ")
}
```

```r
suppressPackageStartupMessages({ library(jsonlite); library(filelock) })

db_append <- function(path, record) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  line <- jsonlite::toJSON(record, auto_unbox = TRUE, null = "null", na = "null")
  con <- file(path, open = "at")
  on.exit(close(con))
  cat(line, "\n", sep = "", file = con)
  invisible(TRUE)
}

db_read <- function(path) {
  if (!file.exists(path)) return(list())
  lines <- readLines(path, warn = FALSE)
  lines <- lines[nzchar(lines)]
  lapply(lines, jsonlite::fromJSON, simplifyVector = FALSE)
}

db_next_variant_id <- function(path) {
  rows <- db_read(path)
  if (!length(rows)) return("var_00001")
  ids <- vapply(rows, function(r) r$variant_id, character(1))
  nums <- as.integer(sub("^var_", "", ids))
  sprintf("var_%05d", max(nums) + 1L)
}

db_top_k <- function(path, k, metric = "ibs_recovery") {
  rows <- db_read(path)
  passers <- Filter(function(r) isTRUE(r$screen$gate_pass), rows)
  if (!length(passers)) return(list())
  vals <- vapply(passers, function(r) as.numeric(r$screen[[metric]]), numeric(1))
  ord <- order(vals, decreasing = TRUE)
  passers[ord[seq_len(min(k, length(ord)))]]
}

allocate_variant_ids <- function(state_path, lock_path, count) {
  lk <- filelock::lock(lock_path, exclusive = TRUE, timeout = 10000)
  if (is.null(lk)) me_stop("Could not acquire state file lock (%s)", lock_path)
  on.exit(filelock::unlock(lk))
  state <- if (file.exists(state_path) && file.size(state_path) > 0L) {
    jsonlite::fromJSON(paste(readLines(state_path, warn = FALSE), collapse = "\n"),
                        simplifyVector = FALSE)
  } else list()
  last <- as.integer(state$last_variant_id %||% 0L)
  ids <- sprintf("var_%05d", (last + 1L):(last + count))
  state$last_variant_id <- last + count
  tmp <- paste0(state_path, ".tmp")
  writeLines(jsonlite::toJSON(state, auto_unbox = TRUE, pretty = TRUE), tmp)
  file.rename(tmp, state_path)
  ids
}

rewrite_db_atomically <- function(path, mutator) {
  rows <- db_read(path)
  new_rows <- lapply(rows, mutator)
  tmp <- paste0(path, ".tmp")
  con <- file(tmp, open = "wt"); on.exit(try(close(con), silent = TRUE))
  for (r in new_rows) {
    cat(jsonlite::toJSON(r, auto_unbox = TRUE, null = "null", na = "null"),
        "\n", sep = "", file = con)
  }
  close(con); on.exit()
  file.rename(tmp, path)
  invisible(length(new_rows))
}
```

```r
sample_parents <- function(db, n_top = 3, n_random = 1,
                           metric = "ibs_recovery", seed = NULL) {
  if (!length(db)) {
    return(list(parents = list(), fallback_mode = "seed_generation",
                failure_summary = character()))
  }
  passers <- Filter(function(r) isTRUE(r$screen$gate_pass), db)
  if (length(passers)) {
    vals <- vapply(passers, function(r) as.numeric(r$screen[[metric]]), numeric(1))
    ord <- order(vals, decreasing = TRUE)
    top_idx <- ord[seq_len(min(n_top, length(ord)))]
    rest <- setdiff(ord, top_idx)
    if (!is.null(seed)) set.seed(seed)
    rand_idx <- if (length(rest) && n_random > 0)
      sample(rest, min(n_random, length(rest))) else integer(0)
    return(list(parents = passers[c(top_idx, rand_idx)],
                fallback_mode = "normal",
                failure_summary = character()))
  }
  # Zero-gate-passer fallback: top-N by fitness regardless of gate_pass
  vals <- vapply(db, function(r) as.numeric(r$screen[[metric]] %||% -Inf), numeric(1))
  ord <- order(vals, decreasing = TRUE)
  top_idx <- ord[seq_len(min(n_top, length(ord)))]
  rest <- setdiff(ord, top_idx)
  if (!is.null(seed)) set.seed(seed)
  rand_idx <- if (length(rest) && n_random > 0)
    sample(rest, min(n_random, length(rest))) else integer(0)
  failures <- unique(unlist(lapply(db, function(r) r$screen$gate_failures)))
  list(parents = db[c(top_idx, rand_idx)],
       fallback_mode = "gate_failure",
       failure_summary = as.character(failures))
}
```

parse_proposer_response <- function(text, expected_batch) {
  stripped <- sub("^\\s*```(?:json)?\\s*", "", text, perl=TRUE)
  stripped <- sub("\\s*```\\s*$", "", stripped, perl=TRUE)
  stripped <- trimws(stripped)
  parsed <- tryCatch(jsonlite::fromJSON(stripped, simplifyVector = FALSE),
                     error = function(e) me_stop("JSON parse failed: %s", e$message))
  if (!is.list(parsed) || !length(parsed) ||
      !all(vapply(parsed, function(x) "feature_set_expr" %in% names(x), logical(1)))) {

  }

    feature_set_expr = as.character(unlist(item$feature_set_expr)),
    rationale = as.character(item$rationale %||% "")
  ))
}

```r
summarize_results <- function(df) {
  list(
    ibs      = mean(df$ibs, na.rm = TRUE),
    c_index  = mean(df$c_index, na.rm = TRUE),
    abs_bias = mean(abs(df$bias), na.rm = TRUE),
    coverage = mean(df$coverage, na.rm = TRUE)
  )
}

check_scenario_denom_floor <- function(cca, oracle, epsilon) {
  d <- cca$ibs - oracle$ibs
  if (abs(d) < epsilon) {
    me_stop("Scenario non-informative for recovery_IBS: |IBS_CCA - IBS_oracle| = %.4f < epsilon=%.4f. Pick a scenario with more CCA-oracle gap or adjust missingness/censoring.",
            abs(d), epsilon)
  }
  invisible(TRUE)
}

compute_primary_fitness <- function(variant, cca, oracle, fallback) {
  denom <- cca$ibs - oracle$ibs
  if (abs(denom) < 1e-6 || sign(denom) != sign(cca$ibs - variant$ibs)) {
    # per-variant degenerate — use fallback metric
    val <- switch(fallback$fallback_metric,
      neg_ibs_ipw = - variant$ibs,
      me_stop("Unknown fallback metric: %s", fallback$fallback_metric))
    return(list(value = val, source = paste0("fallback_", fallback$fallback_metric)))
  }
  list(value = (cca$ibs - variant$ibs) / denom, source = "primary")
}

.select_gates <- function(gates, tier) {
  Filter(function(g) is.null(g$tier) || g$tier == "both" || g$tier == tier, gates)
}

.gate_ok <- function(value, gate, baselines) {
  if (is.null(value) || !is.finite(value)) return(FALSE)
  switch(gate$op,
    "<"  = value <  gate$threshold,
    "<=" = value <= gate$threshold,
    ">"  = value >  gate$threshold,
    ">=" = {
      th <- if (!is.null(gate$threshold)) gate$threshold else {
        env <- list2env(list(C_CCA = baselines$CCA$c_index,
                             IBS_CCA = baselines$CCA$ibs,
                             IBS_oracle = (baselines$oracle %||% list())$ibs))
        eval(parse(text = gate$expr), envir = env)
      }
      value >= th
    },
    "in" = value >= gate$range[[1L]] && value <= gate$range[[2L]],
    me_stop("Unknown gate op: %s", gate$op)
  )
}

evaluate_gates <- function(variant, baselines, fitness, tier = c("screen","full")) {
  tier <- match.arg(tier)
  active_gates <- .select_gates(fitness$gates, tier)
  failures <- character()
  for (g in active_gates) {
    val <- variant[[g$metric]]
    if (!.gate_ok(val, g, baselines)) {
      failures <- c(failures, sprintf("%s(%s) failed op='%s' value=%.4f",
                                       g$metric, tier, g$op, val %||% NA))
    }
  }
  list(gate_pass = length(failures) == 0L, gate_failures = failures)
}

score_variant <- function(variant_results, baselines, fitness, tier = "screen") {
  v <- summarize_results(variant_results)
  bs <- lapply(baselines, summarize_results)
  prim <- compute_primary_fitness(v, bs$CCA, bs$oracle, fitness$recovery$degenerate_variant_fallback)
  g <- evaluate_gates(v, bs, fitness, tier = tier)
  list(
    ibs_ipw = v$ibs,
    ibs_recovery = prim$value,
    recovery_source = prim$source,
    c_index = v$c_index,
    abs_bias = v$abs_bias,
    coverage = v$coverage,
    gate_pass = g$gate_pass,
    gate_failures = g$gate_failures,
    status = "ok"
  )
}
```

```r
suppressPackageStartupMessages({ library(parallel); library(digest) })

materialize_variant <- function(feature_set, variant_id, cache_dir) {
  dir.create(cache_dir, recursive = TRUE, showWarnings = FALSE)
  path <- file.path(cache_dir, paste0(variant_id, ".rds"))
  saveRDS(feature_set, path)
  path
}

child_seed_for <- function(parent_seed, variant_id) {
  h <- digest::digest(paste0(parent_seed, "/", variant_id),
                      algo = "xxhash32", serialize = FALSE)
  strtoi(h, base = 16L) %% .Machine$integer.max
}

run_variant_locally <- function(target_script, env_vars, results_file,
                                 timeout_sec = 600) {
  env_str <- vapply(names(env_vars),
    function(k) paste0(k, "=", shQuote(as.character(env_vars[[k]]))),
    character(1))
  cmd <- paste(c("env", env_str, "Rscript", shQuote(target_script)),
                collapse = " ")
  t0 <- Sys.time()
  status <- tryCatch(
    system(cmd, intern = FALSE, timeout = timeout_sec),
    error = function(e) list(status = -1L, msg = conditionMessage(e))
  )
  elapsed <- as.numeric(difftime(Sys.time(), t0, units = "secs"))
  if (is.list(status) || (is.numeric(status) && status != 0) ||
      !file.exists(results_file)) {
    return(list(ok = FALSE, status = "eval_error",
                elapsed_sec = elapsed,
                error = if (is.list(status)) status$msg else sprintf("exit=%s",
                  if (is.numeric(status)) status else NA),
                results_file = results_file))
  }
  list(ok = TRUE, status = "ok", elapsed_sec = elapsed,
       results_file = results_file)
}

run_variants_parallel <- function(variant_jobs, n_workers = 5) {
  # Children return records (list), parent serializes them in caller.
  parallel::mclapply(variant_jobs, function(j) {
    set.seed(j$child_seed)
    r <- run_variant_locally(j$target_script, j$env_vars, j$results_file,
                              timeout_sec = j$timeout_sec %||% 600)
    list(variant_id = j$variant_id, feature_set = j$feature_set,
         child_seed = j$child_seed, run = r)
  }, mc.cores = n_workers, mc.preschedule = FALSE, mc.cleanup = TRUE)
}
```

```r
ME_STATE_SCHEMA <- 1L
ME_GEN_PHASES <- c("proposing","validated","evaluating","db_written","generation_complete")

state_init <- function(path, run_id, config_sha, parent_seed) {
  s <- list(
    schema_version = ME_STATE_SCHEMA,
    run_id = run_id,
    config_sha = config_sha,
    parent_seed = as.integer(parent_seed),
    last_variant_id = 0L,
    current_generation = 0L,
    generation_phase = "proposing",
    generation_phase_progress = list(
      proposed = list(), validated = list(),
      evaluated = list(), db_written = list()
    ),
    gate_passers_history = list(),
    early_stop_counter = 0L,
    last_checkpoint_at = format(Sys.time(), "%FT%T%z")
  )
  state_save(path, s)
  invisible(s)
}

state_load <- function(path) {
  if (!file.exists(path)) return(NULL)
  jsonlite::fromJSON(paste(readLines(path, warn = FALSE), collapse = "\n"),
                     simplifyVector = FALSE)
}

state_save <- function(path, s) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  s$last_checkpoint_at <- format(Sys.time(), "%FT%T%z")
  tmp <- paste0(path, ".tmp")
  writeLines(jsonlite::toJSON(s, auto_unbox = TRUE, pretty = TRUE), tmp)
  file.rename(tmp, path)
  invisible(TRUE)
}

state_transition <- function(path, generation_phase = NULL, current_generation = NULL,
                              last_variant_id = NULL, proposed = NULL,
                              validated = NULL, evaluated = NULL, db_written = NULL,
                              gate_passers_this_gen = NULL) {
  s <- state_load(path) %||% me_stop("State file missing: %s", path)
  if (!is.null(generation_phase)) {
    if (!generation_phase %in% ME_GEN_PHASES)
      me_stop("Invalid generation_phase: %s", generation_phase)
    s$generation_phase <- generation_phase
  }
  if (!is.null(current_generation)) s$current_generation <- current_generation
  if (!is.null(last_variant_id)) s$last_variant_id <- last_variant_id
  for (k in c("proposed","validated","evaluated","db_written")) {
    v <- get(k)
    if (!is.null(v)) s$generation_phase_progress[[k]] <- as.list(as.integer(v))
  }
  if (!is.null(gate_passers_this_gen)) {
    s$gate_passers_history <- c(s$gate_passers_history, as.integer(gate_passers_this_gen))
  }
  state_save(path, s)
  invisible(s)
}

state_verify_config_sha <- function(state, expected_sha) {
  if (!identical(state$config_sha, expected_sha)) {
    me_stop("config_sha mismatch — config changed mid-run. Start a new run folder. expected=%s got=%s",
            expected_sha, state$config_sha)
  }
  invisible(TRUE)
}

state_resume_plan <- function(state) {
  phase <- state$generation_phase
  prog <- state$generation_phase_progress
  switch(phase,
    generation_complete = list(action = "start_next_generation"),
    db_written          = list(action = "start_next_generation"),
    evaluating          = list(
      action = "re_dispatch",
      variants_to_dispatch = setdiff(as.integer(unlist(prog$validated)),
                                      as.integer(unlist(prog$evaluated))) |> as.list()
    ),
    validated           = list(
      action = "re_dispatch",
      variants_to_dispatch = prog$validated
    ),
    proposing           = list(action = "re_propose"),
    me_stop("Unknown phase: %s", phase)
  )
}
```

```r
render_template <- function(tpl, subs) {
  for (k in names(subs)) {
    tpl <- gsub(paste0("\\{\\{", k, "\\}\\}"), subs[[k]], tpl, fixed = FALSE)
  }
  unresolved <- regmatches(tpl, regexpr("\\{\\{[A-Z_]+\\}\\}", tpl))
  if (length(unresolved)) {
    me_stop("Unresolved placeholder(s): %s",
            paste(unique(unresolved), collapse=", "))
  }
  tpl
}
```

```r
gate_calibration_check <- function(cca_results, baselines, fitness,
                                    tier = c("screen","full")) {
  tier <- match.arg(tier)
  v <- summarize_results(cca_results)
  bs <- lapply(baselines, summarize_results)
  g <- evaluate_gates(v, bs, fitness, tier = tier)
  list(ok = g$gate_pass, failures = g$gate_failures)
}

run_all_prereq_checks <- function(cfg) {
  reports <- list()
  # IBS patch check
  for (pre in cfg$prerequisites) {
    if (pre$id == "ibs-patch") {
      target <- file.path(dirname(cfg$baselines$oracle$results_file %||% "."),
                          "..", pre$target) |> normalizePath(mustWork = FALSE)
      reports$ibs_patch <- list(
        id = pre$id,
        ok = has_ibs_patch(target, required_version = as.integer(pre$required_version))
      )
    }
    if (pre$id == "baseline-metadata") {
      names_req <- pre$required_for
      oks <- logical(length(names_req))
      miss <- list()
      for (i in seq_along(names_req)) {
        bname <- names_req[[i]]
        meta <- cfg$baselines[[bname]]$meta_file
        expected <- list(
          scenario_name = cfg$evaluator$env_vars$SCENARIO_NAME,
          n_subjects = as.integer(cfg$evaluator$env_vars$N_SUBJECTS),
          n_sims = as.integer(cfg$evaluator$env_vars$N_SIMS),
          dgp_version = "missing-types-v3",
          ibs_patch_version = 1L
        )
        r <- check_baseline_meta(meta, expected)
        oks[i] <- r$ok
        if (!r$ok) miss[[bname]] <- r$mismatches
      }
      reports$baseline_metadata <- list(id = pre$id, ok = all(oks), mismatches = miss)
    }
  }
  reports
}
```

```r
detect_target_scripts <- function(root) {
  f <- list.files(root, pattern = "(method_comparison|_method_).*\\.R$",
                  full.names = TRUE, recursive = FALSE)
  normalizePath(f, mustWork = FALSE)
}

detect_submit_scripts <- function(root) {
  list.files(root, pattern = "^submit_.*\\.sh$", full.names = TRUE)
}

detect_scenarios <- function(submit_path) {
  lines <- readLines(submit_path, warn = FALSE)
  arr <- regmatches(lines, regexec("SCENARIOS=\\(([^)]+)\\)", lines))
  arr_hits <- Filter(function(m) length(m) == 2L, arr)
  if (length(arr_hits)) {
    return(strsplit(trimws(arr_hits[[1L]][[2L]]), "\\s+")[[1L]])
  }
  eq <- regmatches(lines, regexec("SCENARIO_NAME=([A-Za-z0-9_]+)", lines))
  eq_hits <- Filter(function(m) length(m) == 2L, eq)
  if (length(eq_hits)) return(vapply(eq_hits, function(m) m[[2L]], character(1)))
  character()
}

detect_metrics <- function(rds_path) {
  df <- readRDS(rds_path)
  num <- vapply(df, is.numeric, logical(1))
  setdiff(names(df)[num], c("sim","seed","run_id","seed_block"))
}

next_run_dir <- function(evolve_root, run_date, name) {
  base <- sprintf("%s_%s", run_date, name)
  candidate <- file.path(evolve_root, base)
  if (!dir.exists(candidate) && !file.exists(candidate)) return(candidate)
  for (i in 2:99) {
    c2 <- file.path(evolve_root, sprintf("%s-%02d", base, i))
    if (!dir.exists(c2) && !file.exists(c2)) return(c2)
  }
  me_stop("Cannot allocate run dir — 99 collisions exhausted for %s", base)
}

scan_project <- function(subproject_root) {
  list(
    target_scripts = detect_target_scripts(subproject_root),
    submit_scripts = detect_submit_scripts(subproject_root),
    scenarios_seen = unique(unlist(lapply(detect_submit_scripts(subproject_root),
      function(s) tryCatch(detect_scenarios(s), error = function(e) character()))))
  )
}
```

```r
write_promote_manifest <- function(program_db, manifest_path, top_k, scenarios,
                                     metric = "ibs_recovery") {
  top <- db_top_k(program_db, k = top_k, metric = metric)
  if (!length(top)) me_stop("No gate-passing variants to promote — run more generations first")
  variants <- lapply(seq_along(top), function(i) list(
    rank = i,
    variant_id = top[[i]]$variant_id,
    seed_block = top[[i]]$seed_block %||% i,
    feature_set_expr = top[[i]]$feature_set_expr,
    screen_metric = top[[i]]$screen[[metric]]
  ))
  manifest <- list(
    generated_at = format(Sys.time(), "%FT%T%z"),
    top_k = as.integer(top_k),
    metric = metric,
    scenarios = as.list(scenarios),
    variants = variants
  )
  dir.create(dirname(manifest_path), recursive = TRUE, showWarnings = FALSE)
  writeLines(jsonlite::toJSON(manifest, auto_unbox = TRUE, pretty = TRUE),
             manifest_path)
  invisible(manifest)
}

promote_commands <- function(variants_cache, manifest_path, promote_submit,
                              remote_host, remote_path) {
  if (!nzchar(remote_host) || !nzchar(remote_path)) {

            call. = FALSE)
    remote_host <- "<REMOTE_HOST>"
    remote_path <- "<REMOTE_PATH>"
  }
  c(
    "# 1. Sync variants cache, manifest, and promote wrapper to Longleaf:",
    sprintf("rsync -avz %s/ %s:%s/variants/",
            shQuote(variants_cache), remote_host, shQuote(remote_path)),
    sprintf("rsync -avz %s %s:%s/promote_manifest.json",
            shQuote(manifest_path), remote_host, shQuote(remote_path)),
    sprintf("rsync -avz %s %s:%s/%s",
            shQuote(promote_submit), remote_host, shQuote(remote_path),
            basename(promote_submit)),
    "",
    "# 2. SSH and submit the array job:",
    sprintf("ssh %s 'cd %s && sbatch %s'",
            remote_host, shQuote(remote_path), basename(promote_submit))
  )
}
```

```r
.sub_namespace <- function(ns_tpl, variant_id, scenario, seed_block) {
  x <- ns_tpl
  x <- gsub("\\$\\{VARIANT_ID\\}",   variant_id,            x)
  x <- gsub("\\$\\{SCENARIO_NAME\\}", scenario,              x)
  x <- gsub("\\$\\{SEED_BLOCK\\}",   as.character(seed_block), x)
  x
}

find_variant_scenario_rds <- function(manifest, root_pattern) {
  paths <- character()
  for (v in manifest$variants) {
    for (scn in manifest$scenarios) {
      paths <- c(paths,
        .sub_namespace(root_pattern, v$variant_id, scn, v$seed_block))
    }
  }
  paths
}

aggregate_variant_full <- function(variant, scenarios, root_pattern) {
  dfs <- list()
  for (scn in scenarios) {
    p <- .sub_namespace(root_pattern, variant$variant_id, scn, variant$seed_block)
    if (!file.exists(p)) {
      me_log("WARN", "missing full-tier output: %s", p)
      next
    }
    df <- readRDS(p)
    df$scenario <- scn
    df$variant_id <- variant$variant_id
    dfs[[length(dfs) + 1L]] <- df
  }
  if (!length(dfs)) return(NULL)
  do.call(rbind, dfs)
}

ingest_manifest <- function(cfg, manifest_path, results_file = "results.rds") {
  manifest <- jsonlite::fromJSON(paste(readLines(manifest_path), collapse="\n"),
                                  simplifyVector = FALSE)
  root_pattern <- file.path(cfg$compute$full$promote$output_namespace, results_file)
  oracle <- readRDS(cfg$baselines$oracle$results_file)
  cca    <- readRDS(cfg$baselines$CCA$results_file)
  scenarios <- unlist(manifest$scenarios)

  aggs <- list()
  for (v in manifest$variants) {
    agg <- aggregate_variant_full(v, scenarios, root_pattern)
    if (!is.null(agg)) aggs[[v$variant_id]] <- agg
  }

  patched <- 0L
  rewrite_db_atomically(cfg$paths$program_db, function(r) {
    agg <- aggs[[r$variant_id]]
    if (is.null(agg)) return(r)
    sc <- score_variant(agg, list(oracle=oracle, CCA=cca),
                         cfg$fitness, tier = "full")
    r$full <- sc
    patched <<- patched + 1L
    r
  })

  list(patched = patched,
       missing_variants = setdiff(vapply(manifest$variants,
         function(v) v$variant_id, character(1)),
         names(aggs)))
}
```

```r
  )
  for (i in seq_along(top)) {
    r <- top[[i]]
    feats <- paste(unlist(r$feature_set_expr), collapse = ", ")
    lines <- c(lines, sprintf(
      "| %d | %s | %d | %.4f | %.4f | %.4f | %.4f | %s |",
      i, r$variant_id, r$generation, r$screen[[metric]],
      r$screen$c_index, r$screen$abs_bias, r$screen$coverage, feats))
  }
  dir.create(dirname(out_path), recursive = TRUE, showWarnings = FALSE)
  writeLines(lines, out_path)
  invisible(out_path)
}

write_final_report <- function(program_db, out_path, config_name, config_path,
                                metric = "ibs_recovery") {
  rows <- db_read(program_db)
  top <- db_top_k(program_db, k = 1L, metric = metric)
  n_all   <- length(rows)
  n_valid <- sum(vapply(rows, function(r) isTRUE(r$grammar_valid), logical(1)))
  n_pass  <- sum(vapply(rows, function(r) isTRUE(r$screen$gate_pass), logical(1)))

  lines <- c(
    sprintf("# Final report — %s", config_name),
    "",
    sprintf("Config: `%s`", config_path),
    sprintf("Generated: %s", format(Sys.time(), "%FT%T%z")),
    ""
  )
  if (!length(top)) {
    lines <- c(lines, "## No gate-passing variants found.", "",
               "Consider: relaxing coverage range, widening bias threshold,",
               "selecting an easier screen scenario, or running more generations.")
  } else {
    w <- top[[1L]]
    feats <- paste(unlist(w$feature_set_expr), collapse = ", ")
    lines <- c(lines,
      "## Winner", "",
      sprintf("- variant_id: `%s`", w$variant_id),
      sprintf("- generation: %d", w$generation),
      sprintf("- seed_block: %d", w$seed_block %||% NA_integer_),
      sprintf("- %s: **%.4f**", metric, w$screen[[metric]]),
      sprintf("- c_index: %.4f", w$screen$c_index),
      sprintf("- abs_bias: %.4f", w$screen$abs_bias),
      sprintf("- coverage: %.4f", w$screen$coverage),
      sprintf("- feature_set: `%s`", feats)
    )
    if (!is.null(w$full)) {
      lines <- c(lines, "",
        "## Full-tier confirmation", "",
        sprintf("- full.%s: %.4f", metric, w$full[[metric]]),
        sprintf("- full.c_index: %.4f", w$full$c_index),
        sprintf("- full.abs_bias: %.4f", w$full$abs_bias),
        sprintf("- full.coverage: %.4f", w$full$coverage),
        sprintf("- full.gate_pass: %s", w$full$gate_pass))
    }
  }
  lines <- c(lines, "",
    "## Run summary", "",
    sprintf("- variants proposed: %d", n_all),
    sprintf("- grammar-valid: %d (%.1f%%)", n_valid,
            if (n_all > 0) 100 * n_valid / n_all else 0),
    sprintf("- gate-passing: %d (%.1f%%)", n_pass,
            if (n_all > 0) 100 * n_pass / n_all else 0)
  )
  dir.create(dirname(out_path), recursive = TRUE, showWarnings = FALSE)
  writeLines(lines, out_path)
  invisible(out_path)
}

write_plots <- function(program_db, out_dir, metric = "ibs_recovery") {
  rows <- db_read(program_db)
  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)
  if (!length(rows)) return(invisible(NULL))

  # Okabe-Ito
  okabe <- c("#E69F00","#56B4E9","#009E73","#F0E442",
             "#0072B2","#D55E00","#CC79A7","#000000")

  # 1. fitness_by_gen: best primary metric per generation
  by_gen <- split(rows, vapply(rows, function(r) r$generation, integer(1)))
  best <- vapply(by_gen, function(grp) {
    passers <- Filter(function(r) isTRUE(r$screen$gate_pass), grp)
    if (!length(passers)) return(NA_real_)
    max(vapply(passers, function(r) as.numeric(r$screen[[metric]]), numeric(1)))
  }, numeric(1))
  pdf(file.path(out_dir, "fitness_by_gen.pdf"), width = 6, height = 4, bg = "white")
  plot(as.integer(names(by_gen)), best, type = "b", pch = 19, col = okabe[[2L]],
       xlab = "generation", ylab = paste("best", metric, "(gate-passers)"),
       main = "Fitness by generation")
  dev.off()

  # 2. top_feature_frequencies among top-20
  top <- db_top_k(program_db, k = 20L, metric = metric)
  feats <- unlist(lapply(top, function(r) unlist(r$feature_set_expr)))
  if (length(feats)) {
    tab <- sort(table(feats), decreasing = TRUE)[1:min(15, length(table(feats)))]
    pdf(file.path(out_dir, "top_feature_frequencies.pdf"), width = 7, height = 5,
        bg = "white")
    op <- par(mar = c(5, 10, 4, 2))
    barplot(rev(tab), horiz = TRUE, las = 1, col = okabe[[1L]],
            main = "Top feature frequencies (top-20 variants)", xlab = "count")
    par(op)
    dev.off()
  }

  # 3. gate_pass_rate_by_gen
  rate <- vapply(by_gen, function(grp) {
    n <- length(grp); if (!n) return(NA_real_)
    mean(vapply(grp, function(r) isTRUE(r$screen$gate_pass), logical(1)))
  }, numeric(1))
  pdf(file.path(out_dir, "gate_pass_rate_by_gen.pdf"), width = 6, height = 4,
      bg = "white")
  plot(as.integer(names(by_gen)), rate, type = "b", pch = 19, col = okabe[[3L]],
       xlab = "generation", ylab = "gate-pass rate",
       main = "Gate-pass rate by generation", ylim = c(0, 1))
  dev.off()
  invisible(NULL)
}
```
